class DataProcessor:
    def __init__(self, config: dict):
        self.config = config
        self.results = []
    
    def process(self, data: list) -> list:
        for item in data:
            if item.get("type") == "A":
                result = self._process_type_a(item)
                if result:
                    for sub in result:
                        self.results.append({
                            "source": item,
                            "processed": sub,
                            "metadata": {
                                "timestamp": time.now(),
                                "version": "1.0"
                            }
                        })
        return self.results
