### 4. API Integration - TypeScript Types

**Added to `edgequake.ts`** (58 lines total):

```typescript
export interface RebuildKnowledgeGraphRequest {
  new_llm_model?: string;
  new_llm_provider?: string;
  rebuild_embeddings?: boolean;
  force?: boolean;
  reprocess_documents?: boolean;
}

export interface RebuildKnowledgeGraphResponse {
  workspace_id: string;
  status: "graph_cleared" | "completed" | "in_progress";
  nodes_cleared: number;
  edges_cleared: number;
  vectors_cleared: number;
  documents_to_process: number;
  job_id?: string | null;
}

export async function rebuildKnowledgeGraph(
  workspace_id: string,
  options: RebuildKnowledgeGraphRequest = {}
): Promise<RebuildKnowledgeGraphResponse> {
  const { data } = await api.post<RebuildKnowledgeGraphResponse>(
    `/api/v1/workspaces/${workspace_id}/rebuild-knowledge-graph`,
    options
  );
  return data;
}
```
